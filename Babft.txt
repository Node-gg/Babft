local function ensureFolder()

local function cfgPath(name)
    if makefolder and isfolder then return FOLDER .. "/" .. name .. ".json" end
    return FOLDER .. "_" .. name .. ".json"
end
local function sanitizeName(s)
    s = tostring(s or "")
    s = s:gsub("[^%w%- ]", "")
    s = s:gsub("^%s+", "")
    s = s:gsub("%s+$", "")
    return s:sub(1, 30)
end
local function saveMeta()
    if not writefile then return end
    ensureFolder()
    pcall(function() writefile(cfgPath("_settings"), HttpService:JSONEncode(Meta)) end)
end
local function loadMeta()
    if not (isfile and readfile) then return end
    local ok, data = pcall(function()
        local p = cfgPath("_settings")
        if not isfile(p) then return nil end
        return HttpService:JSONDecode(readfile(p))
    end)
    if ok and type(data) == "table" then
        Meta.autosave = data.autosave == true
        Meta.autoload = type(data.autoload) == "string" and data.autoload or nil
        Meta.current = type(data.current) == "string" and data.current or nil
        Meta.configs = type(data.configs) == "table" and data.configs or {}
    end
end
local function listConfigs()
    local names = {}
    if listfiles and isfolder and isfolder(FOLDER) then
        local ok, files = pcall(listfiles, FOLDER)
        if ok and type(files) == "table" then
            for _, f in ipairs(files) do
                local n = tostring(f):match("([^/\\]+)%.json$")
                if n and n:sub(1, 1) ~= "_" then table.insert(names, n) end
            end
        end
    else
        for _, n in ipairs(Meta.configs) do table.insert(names, n) end
    end
    table.sort(names)
    return names
end
local function buildConfigData()
    return {
        FarmSpeed = _G.FarmSpeed,
        HoverHeight = _G.HoverHeight,
        AntiAFK = _G.AntiAFK,
        Theme = _G.ThemeName,
        UISize = _G.UISizeName,
    }
end
local function saveConfigNamed(name)
    if not writefile then return false end
    ensureFolder()
    local ok = pcall(function() writefile(cfgPath(name), HttpService:JSONEncode(buildConfigData())) end)
    if ok then
        if not table.find(Meta.configs, name) then table.insert(Meta.configs, name) end
        saveMeta()
    end
    return ok
end
local function readConfigNamed(name)
    if not (isfile and readfile) then return nil end
    local ok, data = pcall(function()
        local p = cfgPath(name)
        if not isfile(p) then return nil end
        return HttpService:JSONDecode(readfile(p))
    end)
    if ok and type(data) == "table" then return data end
    return nil
end
local function deleteConfigNamed(name)
    pcall(function()
        local p = cfgPath(name)
        if delfile and isfile and isfile(p) then delfile(p) end
    end)
    local i = table.find(Meta.configs, name)
    if i then table.remove(Meta.configs, i) end
    if Meta.autoload == name then Meta.autoload = nil end
    if Meta.current == name then Meta.current = nil end
    saveMeta()
end
local function queueAutoSave()
    if not _G.AutoSave or isLoadingConfig or not Meta.current then return end
    autoSaveToken = autoSaveToken + 1
    local token = autoSaveToken
    task.delay(1, function()
        if token == autoSaveToken and _G.AutoSave and Meta.current then
            saveConfigNamed(Meta.current)
        end
    end)
end

local function makeDraggable(guiObject, dragHandle)
    dragHandle = dragHandle or guiObject
    local dragging, dragInput, dragStart, startPos
    dragHandle.InputBegan:Connect(function(input)
        if input.UserInputType == Enum.UserInputType.MouseButton1 or input.UserInputType == Enum.UserInputType.Touch then
            dragging = true
            dragStart = input.Position
            startPos = guiObject.Position
            input.Changed:Connect(function()
                if input.UserInputState == Enum.UserInputState.End then dragging = false end
            end)
        end
    end)
    dragHandle.InputChanged:Connect(function(input)
        if input.UserInputType == Enum.UserInputType.MouseMovement or input.UserInputType == Enum.UserInputType.Touch then
            dragInput = input
        end
    end)
    UserInputService.InputChanged:Connect(function(input)
        if input == dragInput and dragging then
            local delta = input.Position - dragStart
            guiObject.Position = UDim2.new(startPos.X.Scale, startPos.X.Offset + delta.X, startPos.Y.Scale, startPos.Y.Offset + delta.Y)
        end
    end)
end

        valLabel.Text = tostring(currentValue)
        callback(currentValue)
    end
    track.InputBegan:Connect(function(input)
        if input.UserInputType == Enum.UserInputType.MouseButton1 or input.UserInputType == Enum.UserInputType.Touch then
            sliding = true
            update(input)
        end
    end)
    UserInputService.InputChanged:Connect(function(input)
        if sliding and (input.UserInputType == Enum.UserInputType.MouseMovement or input.UserInputType == Enum.UserInputType.Touch) then
            update(input)
        end
    end)
    UserInputService.InputEnded:Connect(function(input)
        if input.UserInputType == Enum.UserInputType.MouseButton1 or input.UserInputType == Enum.UserInputType.Touch then
            if sliding then
                sliding = false
                queueAutoSave()
            end
        end
    end)
    UIControls[id] = {
        Set = function(val)
            currentValue = val
            fill.Size = UDim2.new(math.clamp((val - min) / (max - min), 0, 1), 0, 1, 0)
            valLabel.Text = tostring(val)
            callback(val)
        end,
        GetValue = function() return currentValue end
    }
end

local Stats = {runs = 0, totalTime = 0, lastTime = 0, activeTime = 0, runStart = nil}

local function getRoot()
    local char = LocalPlayer.Character
    local hum = char and char:FindFirstChildOfClass("Humanoid")
    local root = char and char:FindFirstChild("HumanoidRootPart")
    if root and hum and hum.Health > 0 then return root end
    return nil
end
local function getChest()
    local bs = Workspace:FindFirstChild("BoatStages")
    local ns = bs and bs:FindFirstChild("NormalStages")
    local te = ns and ns:FindFirstChild("TheEnd")
    return te and te:FindFirstChild("GoldenChest")
end
local function getObjPos(obj)
    if obj:IsA("BasePart") then return obj.Position end
    if obj:IsA("Model") then return obj:GetPivot().Position end
    return nil
end
local function getWater()
    local water = Workspace:FindFirstChild("Water")
    if not water then return nil end
    local cf, size
    if water:IsA("Model") then
        cf, size = water:GetBoundingBox()
    elseif water:IsA("BasePart") then
        cf, size = water.CFrame, water.Size
    else
        return nil
    end
    return cf.Position, cf.Position.Y + size.Y / 2
end
local function getHoverY()
    return waterTopY + _G.HoverHeight + HOVER_BIAS
end
stopFarm = function()
    farmToken = farmToken + 1
    runActive = false
    reached = false
    for _, c in ipairs(farmConns) do c:Disconnect() end
    table.clear(farmConns)
    if farmBV then farmBV:Destroy() farmBV = nil end
    local char = LocalPlayer.Character
    if char then
        for _, p in ipairs(char:GetDescendants()) do
            if p:IsA("BasePart") and p.Name ~= "HumanoidRootPart" then
                p.CanCollide = true
            end
        end
    end
end
local function beginRun(root)
    local wpos, topY = getWater()
    if not wpos then return false end
    waterTopY = topY
    root.CFrame = CFrame.new(wpos.X, getHoverY(), wpos.Z)
    root.AssemblyLinearVelocity = Vector3.zero
    reached = false
    runActive = true
    Stats.runStart = tick()
    return true
end
local function noclipStep()
    local char = LocalPlayer.Character
    if not char then return end
    for _, p in ipairs(char:GetDescendants()) do
        if p:IsA("BasePart") then p.CanCollide = false end
    end
end
local function moveStep()
    if not _G.AutoFarm then return end
    local root = getRoot()
    if not root then return end
    if not farmBV or farmBV.Parent ~= root then
        if farmBV then farmBV:Destroy() end
        farmBV = Instance.new("BodyVelocity")
        farmBV.MaxForce = Vector3.new(math.huge, math.huge, math.huge)
        farmBV.Velocity = Vector3.zero
        farmBV.Parent = root
    end
    if not runActive then
        farmBV.Velocity = Vector3.zero
        return
    end
    local hoverY = getHoverY()
    if root.Position.Y < hoverY - 2 then
        root.CFrame = CFrame.new(root.Position.X, hoverY, root.Position.Z) * (root.CFrame - root.CFrame.Position)
    end
    local yFix = Vector3.new(0, math.clamp((hoverY - root.Position.Y) * 8, -30, 60), 0)
    local chest = getChest()
    local cpos = chest and getObjPos(chest)
    if cpos then
        local flat = Vector3.new(cpos.X - root.Position.X, 0, cpos.Z - root.Position.Z)
        if flat.Magnitude <= CHEST_TP_DISTANCE then
            root.CFrame = CFrame.new(cpos + Vector3.new(0, 3, 0))
            farmBV.Velocity = Vector3.zero
            runActive = false
            reached = true
        else
            farmBV.Velocity = flat.Unit * _G.FarmSpeed + yFix
        end
    else
        farmBV.Velocity = FORWARD.Unit * _G.FarmSpeed + yFix
    end
end
local function farmLoop(token)
    local function alive() return _G.AutoFarm and token == farmToken end
    while alive() do
        local root
        while alive() do
            root = getRoot()
            if root then break end
            task.wait(0.25)
        end
        if not alive() then break end
        task.wait(1.5)
        if not alive() then break end
        root = getRoot()
        if root then
            if not beginRun(root) then
                sendNotification("Water not found - Auto Farm stopped")
                if UIControls.AutoFarm then UIControls.AutoFarm.Set(false, true) end
                break
            end
            while alive() and runActive and getRoot() == root do task.wait(0.1) end
            if reached and alive() then
                local dt = tick() - Stats.runStart
                Stats.runs = Stats.runs + 1
                Stats.totalTime = Stats.totalTime + dt
                Stats.lastTime = dt
                sendNotification(string.format("Run #%d complete (%.1fs)", Stats.runs, dt))
                local waited = 0
                while alive() and getRoot() == root and waited < RESPAWN_TIMEOUT do
                    task.wait(0.25)
                    waited = waited + 0.25
                end
            end
            runActive = false
            reached = false
        end
    end
end
local function startFarm()
    stopFarm()
    local token = farmToken
    table.insert(farmConns, RunService.Stepped:Connect(noclipStep))
    table.insert(farmConns, RunService.Heartbeat:Connect(moveStep))
    task.spawn(farmLoop, token)
end
